package app;

import java.util.Locale;
import java.util.Scanner;

public class Caixa {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in).useLocale(Locale.US);
        System.out.println("SISTEMA DE VENDAS - CAIXA ABERTO");

        System.out.print("Deseja abrir o caixa para vendas (s/n): ");
        char abrirCaixa = sc.next().toUpperCase().charAt(0);
        int numeroVenda = 0;

        while (abrirCaixa == 'S') {
            numeroVenda++;
            System.out.println("\n--- Venda número: " + numeroVenda + " ---");

            double totalVenda = 0.0;
            char maisItens;

            // Loop para somar os itens da venda atual
            do {
                System.out.print("Valor do item: ");
                double valorItem = sc.nextDouble();

                System.out.print("Quantidade do item: ");
                double qtde = sc.nextDouble();

                double subTotalItem = valorItem * qtde;
                totalVenda += subTotalItem; // Acumula o valor total da compra

                System.out.println("Subtotal acumulado: " + totalVenda);

                System.out.print("Incluir mais itens? (s/n): ");
                maisItens = sc.next().toLowerCase().charAt(0);

            } while (maisItens == 's');

            // Pagamento (após fechar a lista de itens)
            System.out.println("\nTotal da venda: " + totalVenda);
            System.out.print("Qual o valor pago pelo cliente? ");
            double valorPago = sc.nextDouble();

            if (valorPago < totalVenda) {
                System.out.println("Valor insuficiente! Faltam R$ " + (totalVenda - valorPago));
            } else {
                double troco = valorPago - totalVenda;
                System.out.printf("Troco: %.2f\n", troco);
            }

            // Pergunta se deseja realizar uma nova venda ou fechar o caixa
            System.out.print("\nDeseja realizar outra venda? (s/n): ");
            abrirCaixa = sc.next().toUpperCase().charAt(0);
        }

        System.out.println("Caixa fechado. Até logo!");
        sc.close();
    }
}
