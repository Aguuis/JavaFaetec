package app;

import java.util.Locale;
import java.util.Scanner;

public class CalcularSalario {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in).useLocale(Locale.US);

        System.out.println("Digite a matricula: ");
        int matricula = sc.nextInt();

        System.out.println("Salário base: ");
        float salario = sc.nextFloat();

        System.out.println("Digite o código da função (100, 300, 400, 600)");
        int funcao = sc.nextInt();

        float percentual = 0.0f;

        switch(funcao){
            case 100:
                percentual = 2.0f;
                break;
            case 300:
                percentual = 2.0f;
                break;
            case 400:
                percentual = 5.0f;
                break;
            case 600:
                percentual = 7.0f;
                break;
            default:
                System.out.println("Código da função inválido");
                sc.close();
                return;
        }

        float reajuste = salario / 100 * percentual;
        System.out.println("Seu reajuste foi de " + reajuste);

        float novoSalario = salario + reajuste;
        System.out.println("Novo salario: ");


    }
}
