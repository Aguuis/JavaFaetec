package app;

import java.util.Locale;
import java.util.Scanner;

public class Combustivel {
    public static void main(String[] args) {
        double alcool = 3.9;
        double gasolina = 4.3;
        double desconto = 0;

        Scanner sc = new Scanner(System.in).useLocale(Locale.US);

        System.out.println("gasolina ou alcool? ");
        char combustivel = sc.next().charAt(0);

        if (combustivel == 'g') {
            System.out.println("Quantos litros de gasolina? ");
            float litros = sc.nextFloat();
            if (litros <= 20) {
                double total = (gasolina * litros);
                desconto = total-(total * 0.04);
                System.out.println("Valor a pagar: " + desconto);
            }else{
                double total = (gasolina * litros);
                desconto = total-(total * 0.06);
                System.out.println("Valor a pagar: " + desconto);
            }
        }
            if (combustivel == 'a') {
                System.out.println("Quantos litros de alcool? ");
                float litrosA = sc.nextFloat();
                if (litrosA <= 20) {
                    double total = (alcool * litrosA);
                    desconto = total-(total * 0.03);
                    System.out.println("Valor a pagar: " + desconto);
                }else{
                    double total = (alcool * litrosA);
                    desconto = total-(total * 0.05);
                    System.out.println("Valor a pagar: " + desconto);
                }
            }
    }
}
