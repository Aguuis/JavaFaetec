import java.util.Locale;
import java.util.Scanner;

public class FestaEstranha {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in).useLocale(Locale.US);
        System.out.println("Digite sua idade: ");
        int idade = sc.nextInt();

        System.out.println("Digite seu salário: ");
        int salario = sc.nextInt();

        System.out.println("Você é amigo do dono? (s/n): ");
        char amigo = sc.next().toUpperCase().charAt(0);

        if(idade >= 18 && salario >= 3000 || amigo == 'S') {
            System.out.println("Entra na festa!");
        }else{
            System.out.println("Não entra na festa!");
        }
    }
}
