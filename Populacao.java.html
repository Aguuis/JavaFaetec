package app;

public class Populacao {
    public static void main(String[] args) {
        double paisA = 80000;
        double porcentagemA = 0.03;
        double paisB = 200000;
        double porcentagemB = 0.015;
        int ano = 0;

        System.out.println("olá");

       while(paisA <= paisB){
            paisA = (paisA * porcentagemA) + paisA;
            paisB = (paisB * porcentagemB) + paisB;
            ano++;
           System.out.println("Ano " + ano + " -> País A: " + paisA + " | País B: " + paisB);
        }
        System.out.println("\nO País A ultrapassará o País B em " + ano + " anos.");
    }
}
